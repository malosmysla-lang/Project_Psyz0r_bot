cmake_minimum_required(VERSION 3.20)
project(Psyz0r_bot VERSION 0.2.0 LANGUAGES CXX)
add_library(psyz0r_core src/behavior.cpp)
target_include_directories(psyz0r_core PUBLIC include)
target_compile_features(psyz0r_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(psyz0r_core PUBLIC /EHsc /utf-8 PRIVATE /W4 /permissive-)
else()
  target_compile_options(psyz0r_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(psyz0r_sandbox src/main.cpp)
target_link_libraries(psyz0r_sandbox PRIVATE psyz0r_core)
enable_testing()
add_executable(behavior_tests tests/behavior_tests.cpp)
target_link_libraries(behavior_tests PRIVATE psyz0r_core)
add_test(NAME behavior_scenarios COMMAND behavior_tests)
add_executable(state_tests tests/state_tests.cpp)
target_link_libraries(state_tests PRIVATE psyz0r_core)
add_test(NAME state_transitions COMMAND state_tests)
